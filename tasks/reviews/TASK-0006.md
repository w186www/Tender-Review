version = dict(version_id='v', source_id='s', version_label='v1',
               verification_status='verified', text_sha256='0'*64, provisions_sha256='0'*64)
result = dict(schema_version='v1alpha1', result_id='r', corpus_version='c',
    candidates=[dict(candidate_id='x', provision_id='p', version_id='v', score={'fused': 1},
        decision={'citable': True, 'reason_code': 'version_unverified'},
        evidence={'arbitrary': 'not EvidenceRef'})],
    citable_evidence=['missing-id'], excluded=[], degraded=[])
